  try{

    setSyncStatus("connecting");

    if(!await initFirebase()){ setSyncStatus("error"); toast("⚠️ Paste Firebase config in Settings"); return; }

    const db=firebase.firestore();

    const ref=db.collection("exam_os_rooms").doc(S.sync.room);

    const snap=await ref.get();

    if(snap.exists){

      const remote=snap.data();

      if(remote&&remote.data){

        if(!S._lastSync){

          /* first time on this device: cloud already has data */

          openConfirm("Cloud already has data for this room. Replace THIS device's data with cloud? (Cancel = keep local, cloud gets overwritten)",()=>{

            _applyingRemote=true;

            const incoming=remote.data;

            Object.keys(incoming).forEach(k=>{ S[k]=incoming[k]; });

            S._lastSync=remote.updatedAt||new Date().toISOString();

            S.sync={enabled:true,room:ref.id};

            _applyingRemote=false;

            save(); rerender(); toast("⬇ Pulled cloud data");

          });

        } else if(remote.updatedAt>S._lastSync){

          _applyingRemote=true;

          const incoming=remote.data;

          Object.keys(incoming).forEach(k=>{ S[k]=incoming[k]; });

          S._lastSync=remote.updatedAt;

          S.sync={enabled:true,room:ref.id};

          _applyingRemote=false;

          save(); rerender(); toast("⬇ Updated from cloud");

        }

      }

    } else {

      await pushNow(true);

    }

    if(syncUnsub) syncUnsub();

    syncUnsub=ref.onSnapshot(doc=>{

      if(!doc.exists||_applyingRemote) return;

      const d=doc.data(); if(!d||!d.data) return;

      if(d.updatedAt&&(!S._lastSync||d.updatedAt>S._lastSync)){

        _applyingRemote=true;

        store.set(KEY,JSON.stringify(S));

      }

    });

    setSyncStatus("synced");

  }catch(e){ setSyncStatus("error"); console.error(e); toast("⚠️ Sync error",e.message); }
